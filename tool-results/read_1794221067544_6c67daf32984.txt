     1→#!/usr/bin/env bash
     2→# ============================================================================
     3→# clearndr-update.sh — обновление установленного Clear NDR Community
     4→# (docker-стек под управлением stamusctl) на новую версию с сохранением
     5→# всех настроек, включая интеграции (mikrocata2selks / MikroTik и любые другие).
     6→#
     7→# Основано на официальной документации Stamus Networks (Clear NDR):
     8→#   https://docs.clearndr.io/docs/start/common-operations   (upgrade-процедура)
     9→#   https://docs.clearndr.io/docs/stamusctl/compose/update  (команда update)
    10→#   https://github.com/StamusNetworks/stamusctl             (CLI)
    11→#
    12→# Процедура штатного обновления, которой следует скрипт:
    13→#   stamusctl compose update [--version VER]   # новый шаблон + сохранение значений
    14→#   stamusctl compose pull                     # свежие образы
    15→#   stamusctl compose up -d                    # перезапуск стека
    16→#
    17→# Возможности (v3):
    18→#   • Статус каждого долгого шага: «⏳ подпись (сек)» на ОДНОЙ строке терминала,
    19→#     в неинтерактивном режиме — строки [WAIT] раз в минуту; итог «готово за Nс».
    20→#   • Чистый вывод и лог: кадры прогресса docker (Downloading/Extracting/Waiting)
    21→#     перезаписывают одну строку на экране и НЕ попадают в лог — лог содержит
    22→#     только начало/завершение шагов, статусы и ошибки (без 80 МБ прогресса).
    23→#   • Анализ совместимости данных ДО остановки стека: если компонент меняет
    24→#     мажорную версию (OpenSearch 2.x → 3.x, PostgreSQL 16 → 17), скрипт СПРАШИВАЕТ:
    25→#     сохранить данные / ручная конвертация (подсказки) / чистые данные
    26→#     (для PostgreSQL — автоматическое пересоздание тома + восстановление из дампа).
    27→#   • Сохранение всех настроек (3 уровня бэкапа):
    28→#       1. pg_dumpall PostgreSQL ДО остановки (правила, наборы, SID, токены…)
    29→#       2. каталог config/ ПОСЛЕ остановки (compose-файлы, TLS, значения stamusctl);
    30→#          пользовательские файлы, исчезнувшие при обновлении шаблона,
    31→#          восстанавливаются автоматически
    32→#       3. docker volume db-data-<seed> (сырые файлы PostgreSQL)
    33→#   • Интеграции: обнаружение mikrocata2selks (MikroTik), снапшот её файлов,
    34→#     проверка/восстановление suricata.interfaces (tzsp0 и др.), пост-проверка
    35→#     служб/интерфейсов; универсальный механизм сохранения кастомизаций.
    36→#   • Health-checks (контейнеры, веб-интерфейс, REST API по токену), сбор
    37→#     диагностики при сбое (fail-logs) и автооткат config/ с проверкой результата.
    38→#
    39→# Примеры:
    40→#   sudo ./clearndr-update.sh                              # до последней версии
    41→#   sudo ./clearndr-update.sh --version 1.2.0              # до конкретной версии
    42→#   ./clearndr-update.sh --dry-run                         # план без изменений
    43→#   sudo ./clearndr-update.sh --yes --update-cli           # + обновить stamusctl
    44→#   sudo ./clearndr-update.sh --instance /opt/ClearNDR     # не из каталога инстанса
    45→#   sudo ./clearndr-update.sh --opensearch-data wipe       # события OpenSearch очистить
    46→#   sudo ./clearndr-update.sh --pg-strategy restore        # PG major: пересоздать и восстановить
    47→#
    48→# Коды выхода: 0 — успех/отмена без изменений; 1 — ошибка предпроверки;
    49→# 2 — неудача обновления (после попытки отката).

   123→# ----------------------------- статус долгих шагов --------------------------
   124→# Тикер: на терминале обновляет ОДНУ строку «⏳ подпись (сек)»; в неинтерактивном
   125→# режиме пишет в лог строку [WAIT] раз в HEARTBEAT секунд.
   126→TICK_PID=""; TICK_START=0; TICK_LABEL=""
   127→
   128→tick_start() {
   129→  TICK_START=$SECONDS
   130→  TICK_LABEL="$1"
   131→  (
   132→    while :; do
   133→      sleep 3
   134→      local el=$((SECONDS - TICK_START))
   135→      if [ "$IS_TTY" -eq 1 ] && [ "$HAS_TTY" -eq 1 ]; then
   136→        printf '\r\033[K  ⏳ %s… (%ds)' "$TICK_LABEL" "$el" > /dev/tty 2>/dev/null || true
   137→      elif [ $((el % HEARTBEAT)) -eq 0 ]; then
   138→        printf '[WAIT] %s… (%ds)\n' "$TICK_LABEL" "$el"
   139→      fi
   140→    done
   141→  ) & TICK_PID=$!
   142→}
   143→
   144→tick_stop() {
   145→  local rc="${1:-0}"
   146→  if [ -n "$TICK_PID" ]; then
   147→    kill "$TICK_PID" 2>/dev/null || true
   148→    wait "$TICK_PID" 2>/dev/null || true
   149→    TICK_PID=""
   150→  fi
   151→  local el=$((SECONDS - TICK_START))
   152→  if [ "$IS_TTY" -eq 1 ] && [ "$HAS_TTY" -eq 1 ]; then
   153→    printf '\r\033[K' > /dev/tty 2>/dev/null || true
   154→  fi
   155→  if [ "$rc" -eq 0 ]; then
   156→    msg_ok "$TICK_LABEL — готово (за ${el}с)"
   157→  else
   158→    msg_err "$TICK_LABEL — ошибка (за ${el}с)"
   159→  fi
   160→}

   162→# Фильтр вывода docker/stamusctl: кадры прогресса (Downloading/Extracting/Waiting/…)
   163→# на терминале перезаписывают одну строку, а в лог НЕ попадают вовсе. Значимые
   164→# события (Pulled/Pull complete/Created/Started/Healthy/ошибки/warnings) печатаются.
   165→filter_prog() {
   166→  awk -v MODE="$FILTER_MODE" '
   167→    {
   168→      line = $0
   169→      if (line ~ /Downloading \[|Extracting [0-9]|Verifying Checksum|Pulling fs layers/) {
   170→        if (MODE == "tty") { printf "\r\033[K%.100s", line > "/dev/tty"; pending = 1 }
   171→        next
   172→      }
   173→      if (line ~ /(Waiting|Creating|Starting|Stopping|Removing) *$/) {
   174→        if (MODE == "tty") { printf "\r\033[K%.100s", line > "/dev/tty"; pending = 1 }
   175→        next
   176→      }
   177→      if (pending) { if (MODE == "tty") printf "\n" > "/dev/tty"; pending = 0 }
   178→      print
   179→    }
   180→    END { if (pending && MODE == "tty") printf "\n" > "/dev/tty" }
   181→  '
   182→}
   183→
   184→# Выполнить команду с тикером статуса и фильтрацией вывода. Возвращает код команды.
   185→run_prog() {
   186→  local label="$1"; shift
   187→  local rc=0
   188→  tick_start "$label"
   189→  "$@" 2>&1 | filter_prog || rc="${PIPESTATUS[0]}"
   190→  tick_stop "$rc"
   191→  return "$rc"
   192→}

   194→# Выполнить мутирующую команду (с учётом --dry-run) под тикером.
   195→mut_prog() {
   196→  local label="$1"; shift

   198→    msg_info "dry-run: $*"
   199→    return 0
   200→  fi
   201→  run_prog "$label" "$@"
   202→}

   243→resolve_stamusctl() {
   244→  if [ -n "${STAMUSCTL_PATH:-}" ] && [ -x "$STAMUSCTL_PATH" ]; then
   245→    STAMUSCTL_BIN="$STAMUSCTL_PATH"
   246→  elif command -v stamusctl >/dev/null 2>&1; then
   247→    STAMUSCTL_BIN="$(command -v stamusctl)"
   248→  else
   249→    msg_err "Не найден stamusctl — это основной инструмент управления Clear NDR."
   250→    msg_err "Установить:"
   251→    msg_err "  wget https://dl.clearndr.io/stamusctl-linux-amd64"
   252→    msg_err "  chmod +x stamusctl-linux-amd64 && sudo mv stamusctl-linux-amd64 /usr/local/bin/stamusctl"
   253→    msg_err "или запустите скрипт с переменной STAMUSCTL_PATH=/путь/к/stamusctl"
   254→    return 1
   255→  fi
   256→  msg_ok "stamusctl: $STAMUSCTL_BIN ($("$STAMUSCTL_BIN" version 2>/dev/null | head -1 || echo 'версия не определена'))"
   257→}

   259→find_instance() {
   260→  if [ -n "$CONFIG_DIR_OPT" ]; then
   261→    CONFIG_DIR="$CONFIG_DIR_OPT"
   262→    [ -n "$INSTANCE_DIR" ] || INSTANCE_DIR="$(dirname "$CONFIG_DIR_OPT")"
   263→  else
   264→    [ -n "$INSTANCE_DIR" ] || INSTANCE_DIR="$PWD"
   265→    CONFIG_DIR="$INSTANCE_DIR/config"
   266→  fi
   267→  if [ ! -d "$CONFIG_DIR" ]; then
   268→    msg_err "Каталог конфигурации инстанса не найден: $CONFIG_DIR"
   269→    msg_err "Запустите скрипт из каталога инстанса (родитель config/) или укажите --instance / --config-dir."
   270→    msg_err "Список инстансов на этом хосте: stamusctl config list"
   271→    return 1
   272→  fi
   273→  msg_ok "Каталог конфигурации: $CONFIG_DIR"
   274→}

   285→# Имена контейнеров Clear NDR: "<base-name>-<service>-<seed>" (например
   286→# "config-scirius-FNISvONwp2TtteCD"); seed — суффикс шаблона, он же в volumes.
   287→detect_seed() {
   288→  local ct
   289→  ct="$(docker ps -a --format '{{.Names}}' 2>/dev/null | grep -E -- '-scirius-[A-Za-z0-9_]+$' | head -1 || true)"
   290→  printf '%s' "${ct##*-scirius-}"
   291→}

   293→find_db_volume() {  # имя docker volume с данными PostgreSQL (db-data-<seed>)
   294→  local seed cand
   295→  seed="$(detect_seed)"
   296→  if [ -n "$seed" ]; then
   297→    local exact
   298→    exact="$(docker volume ls --format '{{.Name}}' 2>/dev/null | grep -Fx -- "db-data-$seed" || true)"
   299→    if [ -n "$exact" ]; then echo "$exact"; return 0; fi
   300→  fi
   301→  while IFS= read -r cand; do
   302→    [ -n "$cand" ] || continue
   303→    echo "$cand"
   304→    return 0
   305→  done < <(docker volume ls --format '{{.Name}}' 2>/dev/null | grep -E -- '^db-data-[A-Za-z0-9_]+$' || true)
   306→  local only
   307→  only="$(docker volume ls --format '{{.Name}}' 2>/dev/null | grep -E -- 'db-data' || true)"
   308→  if [ "$(echo "$only" | grep -c .)" -eq 1 ]; then
   309→    msg_warn "Volume стека по имени не найден, использую единственный db-data: $only"
   310→    echo "$only"
   311→    return 0
   312→  fi
   313→  return 1
   314→}

   335→# Тег образа компонента из compose-файлов каталога конфигурации.
   336→img_tag_from_config() {  # $1 каталог, $2 подстрока имени образа (opensearch|postgres)
   337→  grep -rhoE "image:[[:space:]]*[\"']?[^\"'[:space:]]*$2:[0-9][^\"'[:space:]]*" "$1" 2>/dev/null \
   338→    | awk '{print $2}' | tr -d "\"'" | head -1

   342→# ----------------------------- шаги процедуры -------------------------------
   343→preflight() {
   344→  step "Проверка окружения"
   345→  need_cmd docker; need_cmd curl; need_cmd tar; need_cmd gzip; need_cmd sha256sum; need_cmd date
   346→  resolve_stamusctl
   347→  find_instance
   348→  docker ps >/dev/null 2>&1 || { msg_err "Docker недоступен (docker ps) — нужен запущенный Docker Engine."; exit 1; }
   349→  msg_ok "Docker Engine доступен"
   350→
   351→  # хост-предусловия OpenSearch (bootstrap-проверки внутри контейнера)
   352→  local mm
   353→  mm="$(sysctl -n vm.max_map_count 2>/dev/null || echo 0)"
   354→  if [ "${mm:-0}" -lt 262144 ]; then
   355→    msg_err "vm.max_map_count = ${mm:-0}, а OpenSearch требует >= 262144 — контейнер opensearch не стартует."
   356→    msg_err "Исправить и сделать постоянным:"
   357→    msg_err "  sudo sysctl -w vm.max_map_count=262144"
   358→    msg_err "  echo 'vm.max_map_count=262144' | sudo tee /etc/sysctl.d/99-clearndr.conf"
   359→    exit 1
   360→  fi
   361→  msg_ok "vm.max_map_count = $mm (>= 262144)"
   362→
   363→  local avail_mb droot dfree
   364→  avail_mb="$(awk '/MemAvailable/ {print int($2/1024)}' /proc/meminfo 2>/dev/null || echo 0)"
   365→  if [ "$avail_mb" -gt 0 ] && [ "$avail_mb" -lt 3072 ]; then
   366→    msg_warn "Доступно памяти ${avail_mb} МБ — OpenSearch по шаблону просит heap 2 ГБ, при нехватке контейнер будет OOM-kill."
   367→    msg_warn "Если обновление упадёт на opensearch: stamusctl config set opensearch.memory=1g && stamusctl compose up -d"
   368→  fi
   369→  droot="$(docker info --format '{{.DockerRootDir}}' 2>/dev/null || echo /var/lib/docker)"
   370→  dfree="$(disk_free_kb "$droot")"
   371→  if [ "${dfree:-0}" -lt $((5 * 1024 * 1024)) ]; then
   372→    msg_err "Мало места на docker-хранилище ($droot): свободно $((dfree/1024)) МБ, нужно >= 5 ГБ (новые образы + работа OpenSearch)."
   373→    exit 1
   374→  fi
   375→  msg_ok "Docker root $droot: свободно $((dfree/1024)) МБ"
   376→
   377→  # текущая версия конфигурации/шаблона
   378→  OLD_STACK_VERSION="$(stack config version 2>/dev/null | grep -Eo '[0-9]+(\.[0-9]+){2}[^[:space:]]*' | tail -1 || true)"
   379→  [ -n "$OLD_STACK_VERSION" ] || OLD_STACK_VERSION="неизвестна"
   380→  msg_ok "Текущая версия стека (config version): $OLD_STACK_VERSION"

   447→  for f in /etc/systemd/system/mikrocata*.service; do [ -e "$f" ] && files+=("$f"); done

   458→    chmod 600 "$out"

   488→postcheck_mikrocata() {  # после обновления: службы, интерфейсы, suricata
   489→  [ "$MIKROCATA" -eq 1 ] || return 0
   490→  [ "$SKIP_MIKROCATA_CHECK" -eq 1 ] && { msg_info "проверка интеграции пропущена (--skip-mikrocata-check)"; return 0; }
   491→  step "Проверка интеграции MikroTik (mikrocata2selks)"
   492→  local bad=0 svc st ifn suri_ct tzsp_hits
   493→  while read -r svc _; do
   494→    [ -n "$svc" ] || continue
   495→    st="$(systemctl is-active "$svc" 2>/dev/null || true)"
   496→    if [ "$st" = "active" ]; then
   497→      msg_ok "служба $svc: active"
   498→    else
   499→      msg_warn "служба $svc: ${st:-не найдена} — docker-обновление её не трогало; проверьте: systemctl status $svc"
   500→      bad=1
   501→    fi
   502→  done < <(systemctl list-unit-files --type=service --no-legend 2>/dev/null | awk '/^mikrocata/ {print $1}')
   503→  local f
   504→  for f in /etc/systemd/network/tzsp*.netdev; do
   505→    [ -e "$f" ] || continue
   506→    ifn="$(basename "$f" .netdev)"
   507→    if ip link show "$ifn" >/dev/null 2>&1; then
   508→      msg_ok "интерфейс $ifn: существует"
   509→    else
   510→      msg_warn "интерфейс $ifn: ОТСУТСТВУЕТ (systemd-networkd?) — проверьте: networkctl status $ifn"
   511→      bad=1
   512→    fi
   513→  done
   514→  suri_ct="$(docker ps --format '{{.Names}}' 2>/dev/null | grep -E -- '-suricata-[A-Za-z0-9_]+ ' | head -1 | awk '{print $1}' || true)"
   515→  if [ -n "$suri_ct" ]; then
   516→    tzsp_hits="$(docker logs --tail 100 "$suri_ct" 2>&1 | grep -ci tzsp || true)"
   517→    if [ "${tzsp_hits:-0}" -gt 0 ]; then
   518→      msg_ok "suricata работает с интерфейсами tzsp (упоминаний в журнале: $tzsp_hits)"
   519→    else
   520→      msg_warn "в журнале suricata нет упоминаний tzsp — если MikroTik-трафик пропал, проверьте: tcpdump -i tzsp0 и journalctl -u mikrocataTZSP0.service"
   521→    fi
   522→  fi
   523→  if [ "$bad" -eq 0 ]; then
   524→    msg_ok "Интеграция MikroTik: служба(ы)/интерфейсы на месте, suricata запущен"
   525→  else
   526→    msg_warn "Есть замечания по интеграции (выше). Docker-обновление Clear NDR файлы mikrocata не меняет."
   527→  fi
   528→  return 0
   529→}

   543→    echo "old postgres:   $OLD_PG_TAG"
   544→    echo "suricata.interfaces: $OLD_SURICATA_INTERFACES"
   545→    echo "mikrocata2selks: $MIKROCATA"

   547→    echo; echo "--- stamusctl config version ---"; stack config version 2>/dev/null || true

   549→    echo; echo "--- docker ps (стек) ---"; docker ps -a --format 'table {{.Names}}\t{{.Image}}\t{{.Status}}' 2>/dev/null | grep -E -- "-(scirius|db|suricata|nginx|opensearch|fluentd|osd|evebox|arkime|utils|rabbitmq|celery|cron|scout|ssl-keygen|setup-|suri-reloader|ownership-)" || echo "(стек не запущен)"

   552→  } > "$tmp" 2>/dev/null || true
   553→  if [ "$DRY_RUN" -eq 1 ]; then

   555→    rm -f "$tmp"
   556→  else
   557→    mv "$tmp" "$MANIFEST" && chmod 600 "$MANIFEST"
   558→    msg_ok "Манифест состояния сохранён: $MANIFEST"
   559→  fi

   574→db_dump() {  # уровень 1: дамп PostgreSQL, пока стек жив
   575→  [ "$SKIP_DB_DUMP" -eq 1 ] && { msg_warn "Дамп БД пропущен (--skip-db-dump)"; return 0; }
   576→  step "Бэкап 1/3: дамп PostgreSQL (настройки правил, пользователи, токены)"
   577→  local dbcid out
   578→  out="$BACKUP_DIR/db-dump-$TS.sql.gz"
   579→  dbcid="$(stack compose ps -q db 2>/dev/null | head -1 || true)"
   580→  if [ -z "$dbcid" ]; then
   581→    dbcid="$(docker ps --format '{{.ID}} {{.Names}}' 2>/dev/null | grep -- '-db' | head -1 | awk '{print $1}' || true)"
   582→  fi
   583→  if [ -z "$dbcid" ]; then
   584→    if stack_running; then
   585→      msg_err "Стек запущен, но контейнер БД (db) не найден — дамп невозможен."
   586→      exit 1
   587→    fi
   588→    msg_warn "Стек не запущен — дамп БД пропущен (настройки правил можно будет проверить в UI)."
   589→    return 0
   590→  fi
   591→  DBUSER="$(docker exec "$dbcid" env 2>/dev/null | sed -n 's/^POSTGRES_USER=//p' | head -1)"
   592→  DBUSER="${DBUSER:-selks}"
   593→  if [ "$DRY_RUN" -eq 1 ]; then
   594→    msg_info "dry-run: docker exec <db> pg_dumpall -U $DBUSER | gzip > $out"
   595→    return 0
   596→  fi
   597→  tick_start "Дамп PostgreSQL (pg_dumpall -U $DBUSER)"
   598→  local drc=0
   599→  docker exec "$dbcid" pg_dumpall -U "$DBUSER" 2>"$BACKUP_DIR/db-dump-$TS.err" | gzip > "$out" || drc="${PIPESTATUS[0]}"
   600→  tick_stop "$drc"
   601→  if [ "$drc" -eq 0 ]; then
   602→    chmod 600 "$out"
   603→    local rows; rows="$(zcat "$out" | grep -c 'CREATE TABLE' || true)"
   604→    msg_ok "Дамп БД: $out ($(du -h "$out" | awk '{print $1}'), таблиц: ${rows:-0}, пользователь: $DBUSER)"
   605→    if [ -s "$BACKUP_DIR/db-dump-$TS.err" ]; then
   606→      msg_warn "Замечания pg_dumpall: $BACKUP_DIR/db-dump-$TS.err"
   607→    fi
   608→  else
   609→    msg_err "Не удалось снять дамп БД — подробности: $BACKUP_DIR/db-dump-$TS.err"
   610→    msg_err "Продолжение опасно: главный носитель настроек не сохранён."
   611→    exit 1
   612→  fi
   613→}

   615→# ----------------------------- решение по данным ----------------------------
   616→print_os_convert_guide() {
   617→  cat <<'EOF'
   618→  Ручная конвертация данных OpenSearch между мажорными версиями (шаги):
   619→    1. Снапшот со старой версии: настроить repository FS (path.repo в opensearch.yml +
   620→       том для снапшотов), затем PUT _snapshot/backup и создать снапшот индексов.
   621→    2. Поднять стек с новой версией, RESTORE снапшота и переиндексировать (reindex)
   622→       индексы, созданные старой версией (OpenSearch 3.x не читает индексы < 2.0.0).
   623→    3. Альтернатива: промежуточный шаблон с OpenSearch 2.x → дождаться зелёного
   624→       кластера → переиндексация → обновление до 3.x.
   625→    Документация: OpenSearch «Snapshot and restore», «Upgrade to OpenSearch 3.x».
   626→  Обновление прервано БЕЗ изменений: стек продолжает работать в прежней версии.
   627→EOF
   628→}

   630→# Решение о данных — ДО остановки стека (отмена/гид = нулевой простой).
   631→decide_data_strategy() {
   632→  step "Анализ совместимости данных (старая версия → целевой шаблон)"
   633→  local old_m new_m
   634→
   635→  # --- OpenSearch ---
   636→  if [ -n "$OLD_OS_TAG" ] && [ -n "${TARGET_OS_TAG:-}" ]; then
   637→    old_m="$(major_of "$OLD_OS_TAG")"; new_m="$(major_of "$TARGET_OS_TAG")"
   638→    if [ "$old_m" != "$new_m" ]; then
   639→      msg_warn "OpenSearch меняет мажорную версию: $OLD_OS_TAG → $TARGET_OS_TAG"
   640→      msg_warn "Индексы событий старой версии могут быть несовместимы (3.x не читает индексы, созданные < 2.0.0)."
   641→      case "$OPENSEARCH_DATA" in
   642→        ask)
   643→          if [ "$ASSUME_YES" -eq 0 ] && [ -t 0 ]; then
   644→            cat <<'EOF'
   645→  Что делать с данными событий OpenSearch?
   646→    1) сохранить данные и продолжить (при несовместимости — автооткат, журналы в fail-logs)
   647→    2) показать шаги ручной конвертации (снапшот/переиндексация) и прервать обновление
   648→    3) обновиться с чистыми данными событий (том сохраняется в tar; настройки правил в PostgreSQL НЕ трогаются)
   649→EOF
   650→            local ans
   651→            read -r -p "Выберите 1/2/3 [1]: " ans || ans="1"; ans="${ans:-1}"
   652→            case "$ans" in
   653→              2) print_os_convert_guide; exit 0 ;;
   654→              3) WIPE_OS=1; msg_ok "Решение: обновиться с чистыми данными событий" ;;
   655→              *) msg_ok "Решение: сохранить данные (автооткат при несовместимости)" ;;
   656→            esac
   657→          else
   658→            msg_warn "Неинтерактивный режим: данные сохраняются по умолчанию (для очистки: --opensearch-data wipe)"
   659→          fi ;;
   660→        wipe)  WIPE_OS=1; msg_ok "Решение (--opensearch-data wipe): чистые данные событий, том в tar" ;;
   661→        guide) print_os_convert_guide; exit 0 ;;
   662→        keep)  msg_ok "Решение (--opensearch-data keep): сохранить данные" ;;
   663→        *) msg_err "--opensearch-data: допустимо keep|wipe|guide|ask"; exit 1 ;;
   664→      esac
   665→      OS_DECIDED=1
   666→    else
   667→      msg_ok "OpenSearch: $OLD_OS_TAG → $TARGET_OS_TAG — мажорная версия не меняется, данные совместимы"
   668→    fi
   669→  else
   670→    msg_warn "Версии OpenSearch определить не удалось (шаблон недоступен?) — решение будет принято после compose update"
   671→  fi
   672→
   673→  # --- PostgreSQL ---
   674→  if [ -n "$OLD_PG_TAG" ] && [ -n "${TARGET_PG_TAG:-}" ]; then
   675→    old_m="$(major_of "$OLD_PG_TAG")"; new_m="$(major_of "$TARGET_PG_TAG")"
   676→    if [ "$old_m" != "$new_m" ]; then
   677→      msg_warn "PostgreSQL меняет мажорную версию: $OLD_PG_TAG → $TARGET_PG_TAG"
   678→      msg_warn "Старый том БД несовместим с новой версией — требуется пересоздание + восстановление из дампа."
   679→      case "$PG_STRATEGY" in
   680→        ask)
   681→          if [ "$ASSUME_YES" -eq 0 ] && [ -t 0 ]; then
   682→            cat <<'EOF'
   683→  Что делать с данными PostgreSQL (настройки правил, пользователи, токены)?
   684→    1) автоматически: пересоздать том и восстановить всё из дампа (рекомендуется; дамп уже снят)
   685→    2) отменить обновление
   686→EOF
   687→            local ans
   688→            read -r -p "Выберите 1/2 [1]: " ans || ans="1"; ans="${ans:-1}"
   689→            case "$ans" in
   690→              2) msg_info "Обновление отменено пользователем (несовместимость PostgreSQL). Стек не тронут."; exit 0 ;;
   691→              *) RESTORE_DB=1; msg_ok "Решение: пересоздать том БД и восстановить из дампа" ;;
   692→            esac
   693→          else
   694→            msg_warn "Неинтерактивный режим: том БД будет пересоздан и восстановлен из дампа (единственный рабочий путь при смене мажорной версии)"
   695→            RESTORE_DB=1
   696→          fi ;;
   697→        restore) RESTORE_DB=1; msg_ok "Решение (--pg-strategy restore): пересоздать том и восстановить дамп" ;;
   698→        cancel)  msg_info "Обновление отменено (--pg-strategy cancel). Стек не тронут."; exit 0 ;;
   699→        *) msg_err "--pg-strategy: допустимо restore|cancel|ask"; exit 1 ;;
   700→      esac
   701→      PG_DECIDED=1
   702→    else
   703→      msg_ok "PostgreSQL: $OLD_PG_TAG → $TARGET_PG_TAG — мажорная версия не меняется, том совместим"
   704→    fi
   705→  fi
   706→  return 0
   707→}

   709→# Повторная проверка ПО ФАКТИЧЕСКОЙ новой конфигурации (если целевые теги
   710→# не удалось получить до обновления). Отмена здесь = восстановление config/.
   711→recheck_strategy_after_update() {
   712→  NEW_OS_TAG="$(img_tag_from_config "$CONFIG_DIR" opensearch || true)"
   713→  NEW_PG_TAG="$(img_tag_from_config "$CONFIG_DIR" postgres || true)"
   714→  local old_m new_m
   715→  if [ "$OS_DECIDED" -eq 0 ] && [ -n "$OLD_OS_TAG" ] && [ -n "${NEW_OS_TAG:-}" ]; then
   716→    old_m="$(major_of "$OLD_OS_TAG")"; new_m="$(major_of "$NEW_OS_TAG")"
   717→    if [ "$old_m" != "$new_m" ]; then
   718→      msg_warn "Подтверждено по новой конфигурации: OpenSearch $OLD_OS_TAG → $NEW_OS_TAG (мажорная версия)"
   719→      case "$OPENSEARCH_DATA" in
   720→        wipe) WIPE_OS=1 ;;
   721→        guide) print_os_convert_guide; cancel_and_restore ;;
   722→        *)
   723→          if [ "$ASSUME_YES" -eq 0 ] && [ -t 0 ]; then
   724→            local ans
   725→            read -r -p "OpenSearch $old_m → $new_m: 1) сохранить данные  2) чистые данные событий  3) отменить обновление [1]: " ans || ans="1"; ans="${ans:-1}"
   726→            case "$ans" in
   727→              2) WIPE_OS=1 ;;
   728→              3) cancel_and_restore ;;
   729→              *) msg_ok "данные сохраняются" ;;
   730→            esac
   731→          else
   732→            msg_warn "Неинтерактивный режим: данные сохраняются (автооткат при несовместимости)"
   733→          fi ;;
   734→      esac
   735→      OS_DECIDED=1
   736→    fi
   737→  fi
   738→  if [ "$PG_DECIDED" -eq 0 ] && [ -n "$OLD_PG_TAG" ] && [ -n "${NEW_PG_TAG:-}" ]; then
   739→    old_m="$(major_of "$OLD_PG_TAG")"; new_m="$(major_of "$NEW_PG_TAG")"
   740→    if [ "$old_m" != "$new_m" ]; then
   741→      msg_warn "Подтверждено по новой конфигурации: PostgreSQL $OLD_PG_TAG → $NEW_PG_TAG (мажорная версия)"
   742→      case "$PG_STRATEGY" in
   743→        restore) RESTORE_DB=1 ;;
   744→        cancel)  cancel_and_restore ;;
   745→        *)
   746→          if [ "$ASSUME_YES" -eq 0 ] && [ -t 0 ]; then
   747→            local ans
   748→            read -r -p "PostgreSQL $old_m → $new_m: 1) пересоздать том и восстановить дамп  2) отменить [1]: " ans || ans="1"; ans="${ans:-1}"
   749→            case "$ans" in
   750→              2) cancel_and_restore ;;
   751→              *) RESTORE_DB=1 ;;
   752→            esac
   753→          else
   754→            RESTORE_DB=1
   755→          fi ;;
   756→      esac
   757→      PG_DECIDED=1
   758→    fi
   759→  fi
   760→  return 0
   761→}

   763→cancel_and_restore() {
   764→  msg_info "Отмена обновления — восстанавливаю прежнюю конфигурацию…"
   765→  local cfg_archive="$BACKUP_DIR/config-$TS.tar.gz"
   766→  if [ -s "$cfg_archive" ]; then
   767→    stack compose down || true
   768→    rm -rf "$CONFIG_DIR"
   769→    tar xzf "$cfg_archive" -C "$INSTANCE_DIR"
   770→    run_prog "Запуск прежнего стека" stack compose up -d || true
   771→    msg_ok "Обновление отменено, прежняя конфигурация восстановлена и запущена"
   772→  else
   773→    msg_err "Архив config/ не найден ($cfg_archive) — восстановите вручную"
   774→  fi
   775→  exit 0
   776→}

   778→confirm_plan() {
   779→  step "План обновления"
   780→  cat <<EOF
   781→  Инстанс:          $INSTANCE_DIR
   782→  Версия стека:     $OLD_STACK_VERSION  ->  $TARGET_VERSION
   783→  OpenSearch:       ${OLD_OS_TAG:-?} -> ${TARGET_OS_TAG:-?}   (данные: $( [ "$WIPE_OS" -eq 1 ] && echo "ОЧИСТИТЬ (события, том в tar)" || echo "сохранить" ))
   784→  PostgreSQL:       ${OLD_PG_TAG:-?} -> ${TARGET_PG_TAG:-?}   (БД: $( [ "$RESTORE_DB" -eq 1 ] && echo "пересоздать + восстановить из дампа" || echo "сохранить (том)" ))
   785→  Интеграция:       $( [ "$MIKROCATA" -eq 1 ] && echo "MikroTik/mikrocata2selks — снапшот + проверка" || echo "mikrocata2selks не обнаружена" )
   786→  suricata.interfaces: ${OLD_SURICATA_INTERFACES:-—} (будут проверены/восстановлены)
   787→  Бэкапы ->         $BACKUP_DIR  (pg-дамп: $([ $SKIP_DB_DUMP -eq 1 ] && echo НЕТ || echo ДА); config/: ДА; volume: $([ $SKIP_VOLUME -eq 1 ] && echo НЕТ || echo ДА))
   788→  Обновление CLI:   $([ $UPDATE_CLI -eq 1 ] && echo "да (stamusctl)" || echo "нет")
   789→  Проверки после:   контейнеры ${STACK_SERVICES[*]} + $HEALTH_URL $([ -n "$API_TOKEN" ] && echo "+ REST API по токену")
   790→  Даунтайм:         да — стек будет остановлен на время обновления (обычно 2-10 минут)
   791→EOF
   792→  [ "$DRY_RUN" -eq 1 ] && { msg_ok "dry-run: изменений не будет"; return 0; }
   793→  if [ "$ASSUME_YES" -eq 0 ]; then
   794→    if [ ! -t 0 ]; then
   795→      msg_err "Нет интерактивного терминала: подтвердите запуск флагом --yes."
   796→      exit 1
   797→    fi
   798→    local ans
   799→    read -r -p "Продолжить обновление с остановкой стека? [да/нет] " ans || ans="нет"
   800→    case "$ans" in
   801→      да|Д|д|y|Y|yes) ;;
   802→      *) msg_info "Отменено пользователем."; exit 0 ;;
   803→    esac
   804→  else
   805→    msg_info "Флаг --yes: подтверждение пропущено"
   806→  fi
   807→}

   927→# Чистка данных OpenSearch (по решению пользователя): том в tar, затем удаление.
   928→wipe_opensearch_data() {
   929→  [ "$WIPE_OS" -eq 1 ] || return 0
   930→  step "Очистка данных событий OpenSearch (том сохраняется в tar)"
   931→  if [ "$DRY_RUN" -eq 1 ]; then
   932→    msg_info "dry-run: копия и удаление volume $(find_opensearch_volume || echo 'opensearch-data-<seed>')"
   933→    return 0
   934→  fi
   935→  local vol
   936→  vol="$(find_opensearch_volume)" || { msg_warn "Volume opensearch-data не найден — чистить нечего"; return 0; }
   937→  local out="$BACKUP_DIR/opensearch-data-$TS.tar.gz"
   938→  local free_kb need_kb
   939→  need_kb=$(($(dir_size_kb_docker "$vol") + 1024 * 1024))
   940→  free_kb="$(disk_free_kb "$BACKUP_DIR")"
   941→  if [ "${free_kb:-0}" -lt "$need_kb" ]; then
   942→    msg_warn "Недостаточно места для копии volume $vol (~$((need_kb/1024/1024)) ГБ) — очистка БЕЗ копии."
   943→    msg_warn "Продолжить очистку без резервной копии событий? История событий будет потеряна безвозвратно."
   944→    if [ "$ASSUME_YES" -eq 0 ] && [ -t 0 ]; then
   945→      local ans; read -r -p "Продолжить без копии? [нет/да] " ans || ans="нет"
   946→      case "$ans" in да|Д|д|y|Y|yes) ;; *) msg_info "Очистка отменена — данные сохраняются."; WIPE_OS=0; return 0 ;; esac
   947→    else
   948→      msg_warn "--yes: продолжаю без копии (задайте --opensearch-data keep, чтобы сохранить данные)"
   949→    fi
   950→  else
   951→    if ! run_prog "Копия volume $vol" docker run --rm -v "$vol":/src:ro -v "$BACKUP_DIR":/dst alpine tar czf "/dst/$(basename "$out")" -C /src .; then
   952→      msg_warn "Копия volume не удалась — очистка отменена, данные сохраняются."
   953→      WIPE_OS=0; return 0
   954→    fi
   955→    chmod 600 "$out"
   956→    [ -s "$out" ] || { msg_warn "Копия volume пуста — очистка отменена."; WIPE_OS=0; return 0; }
   957→    msg_ok "Копия событий: $out ($(du -h "$out" | awk '{print $1}'))"
   958→  fi
   959→  if run_prog "Удаление тома $vol" docker volume rm "$vol"; then
   960→    msg_ok "Том $vol удалён — OpenSearch создаст свежие индексы (настройки правил в PostgreSQL НЕ затронуты)"
   961→  else
   962→    msg_err "Не удалось удалить том $vol — проверьте, что контейнеры opensearch остановлены."
   963→    return 1
   964→  fi
   965→}

   971→# Восстановление пользовательских файлов config/, исчезнувших при обновлении шаблона

   975→  local arch="$BACKUP_DIR/config-$TS.tar.gz"

   977→  local old_files f restored=0

   980→    [ -n "$f" ] || continue

   984→        restored=$((restored + 1))
   985→      fi
   986→    fi
   987→  done <<< "$old_files"
   988→  if [ "$restored" -gt 0 ]; then
   989→    msg_ok "Восстановлено пользовательских файлов: $restored (проверьте их актуальность в новой версии)"
   990→  else
   991→    msg_ok "Пользовательские файлы config/ сохранены новой конфигурацией"
   992→  fi
   993→  return 0
   994→}

   996→wait_db_ready() {
   997→  local el=0 id
   998→  while [ "$el" -lt 90 ]; do
   999→    id="$(stack compose ps -q db 2>/dev/null | head -1 || true)"
  1000→    if [ -n "$id" ] && docker exec "$id" pg_isready -U "$DBUSER" >/dev/null 2>&1; then
  1001→      return 0
  1002→    fi
  1003→    sleep 3; el=$((el + 3))
  1004→  done
  1005→  return 1
  1006→}

  1008→# Пересоздание тома PostgreSQL + восстановление дампа (путь при смене major PG).
  1009→restore_pg_from_dump() {
  1010→  [ "$RESTORE_DB" -eq 1 ] || return 0
  1011→  step "Пересоздание тома PostgreSQL и восстановление настроек из дампа"
  1012→  if [ "$DRY_RUN" -eq 1 ]; then
  1013→    msg_info "dry-run: удаление volume db-data, compose up -d db, psql < db-dump-$TS.sql.gz"
  1014→    return 0
  1015→  fi
  1016→  local vol
  1017→  vol="$(find_db_volume)" || { msg_err "Volume БД не найден — восстановление невозможно."; return 1; }
  1018→  if ! run_prog "Удаление тома БД $vol" docker volume rm "$vol"; then
  1019→    msg_err "Не удалось удалить том $vol (контейнер db должен быть остановлен)."; return 1
  1020→  fi
  1021→  if ! run_prog "Запуск службы db (новый том)" stack compose up -d db; then
  1022→    msg_err "Не удалось запустить службу db."; return 1
  1023→  fi
  1024→  if ! run_prog "Ожидание готовности PostgreSQL" wait_db_ready; then
  1025→    msg_err "PostgreSQL не стал готов за 90с — подробности: docker logs <db-контейнер>"; return 1
  1026→  fi
  1027→  local dbcid
  1028→  dbcid="$(stack compose ps -q db 2>/dev/null | head -1 || true)"
  1029→  local dump="$BACKUP_DIR/db-dump-$TS.sql.gz"
  1030→  if [ ! -s "$dump" ]; then
  1031→    msg_err "Дамп $dump отсутствует — восстановление невозможно (том БД уже пересоздан!)."
  1032→    msg_err "Настройки можно внести вручную либо откатить том: см. backups/volume-…-$TS.tar.gz"
  1033→    return 1
  1034→  fi
  1035→  if run_prog "Восстановление дампа PostgreSQL" bash -c "zcat '$dump' | docker exec -i '$dbcid' psql -U '$DBUSER' -d postgres"; then
  1036→    msg_ok "Настройки восстановлены в новый PostgreSQL ($NEW_PG_TAG)"
  1037→  else
  1038→    msg_err "Ошибка восстановления дампа — смотрите лог; дамп сохранён: $dump"
  1039→    return 1
  1040→  fi
  1041→  return 0
  1042→}

  1074→      "$el" "${p:-—}" "$h" > /dev/tty 2>/dev/null || true
  1075→  elif [ "$el" -gt 0 ] && [ $((el % HEARTBEAT)) -eq 0 ]; then
  1076→    msg_info "ожидание стека (${el}с): не готовы: ${p:-—}; HTTP $h"
  1077→  fi
  1078→}

  1080→wait_healthy() {
  1081→  step "Проверки здоровья (лимит ${HEALTH_TIMEOUT}с)"
  1082→  local elapsed=0 interval=5 ok http=000 pending=""
  1083→  if [ "$DRY_RUN" -eq 1 ]; then
  1084→    msg_info "dry-run: ожидание готовности сервисов ${STACK_SERVICES[*]} и ответа $HEALTH_URL"
  1085→    return 0
  1086→  fi
  1087→  while [ "$elapsed" -lt "$HEALTH_TIMEOUT" ]; do
  1088→    ok=1; pending=""
  1089→    local s status names
  1090→    names="$(docker ps --format '{{.Names}} {{.Status}}' 2>/dev/null || true)"
  1091→    for s in "${STACK_SERVICES[@]}"; do
  1092→      # имена контейнеров Clear NDR: "<base>-<service>-<seed>", строка: "<имя> Up …"
  1093→      status="$(echo "$names" | grep -E -- "-${s}-[A-Za-z0-9_]+ " | head -1 | awk '{print $2}')"
  1094→      if [ "$status" = "Up" ]; then :; else ok=0; pending+="${s} "; fi
  1095→    done
  1096→    http="$(curl -k -s -o /dev/null -w '%{http_code}' --max-time 10 "$HEALTH_URL" 2>/dev/null || echo 000)"
  1097→    if [ "$ok" -eq 1 ] && { [ "$http" = "200" ] || [ "$http" = "302" ] || [ "$http" = "301" ]; }; then
  1098→      clear_wait_line
  1099→      msg_ok "Все сервисы ${STACK_SERVICES[*]} запущены; веб-интерфейс $HEALTH_URL отвечает (HTTP $http)"
  1100→      check_api_token || return 1
  1101→      show_recent_errors
  1102→      return 0
  1103→    fi
  1104→    render_wait_line "$elapsed" "$pending" "$http"
  1105→    sleep "$interval"; elapsed=$((elapsed + interval))
  1106→  done
  1107→  clear_wait_line
  1108→  msg_err "Стек не вышел на рабочий режим за ${HEALTH_TIMEOUT}с (не готовы: ${pending:-?}; HTTP $HEALTH_URL → $http)."
  1109→  return 1
  1110→}

  1112→check_api_token() {
  1113→  [ -n "$API_TOKEN" ] || { msg_info "Токен не задан — проверка REST API пропущена (можно: --token / env CLEAR_NDR_TOKEN)"; return 0; }
  1114→  local code body
  1115→  body="$(curl -k -s --max-time 15 -w '\n%{http_code}' "$HEALTH_URL/rest/rules/source/" -H "Authorization: Token $API_TOKEN" 2>/dev/null || true)"
  1116→  code="$(echo "$body" | tail -1)"
  1117→  if [ "$code" = "200" ]; then
  1118→    local count
  1119→    count="$(echo "$body" | head -n -1 | sed -n 's/.*"count"[[:space:]]*:[[:space:]]*\([0-9]*\).*/\1/p' | head -1)"
  1120→    msg_ok "REST API отвечает 200; источников правил в базе: ${count:-0} — настройки правил на месте"
  1121→    return 0
  1122→  fi
  1123→  msg_warn "REST API вернул HTTP ${code:-?} (ожидалось 200). Токен мог измениться либо UI ещё поднимается — проверьте вручную."
  1124→  return 0
  1125→}

  1158→# Сохраняет статус и журналы контейнеров ДО отката (compose down удаляет логи).
  1159→capture_failure_diagnostics() {
  1160→  [ "$DRY_RUN" -eq 1 ] && return 0
  1161→  step "Диагностика сбоя: сохранение состояния и журналов контейнеров"
  1162→  local diag_dir="$BACKUP_DIR/fail-logs-$TS"
  1163→  mkdir -p "$diag_dir"
  1164→  {
  1165→    docker ps -a --format '{{.Names}}\t{{.Status}}\t{{.Image}}' 2>/dev/null \
  1166→      | grep -E -- '-(scirius|db|suricata|nginx|opensearch|fluentd|osd|evebox|arkime|rabbitmq|suri-reloader|socat|utils|celery|cron|scout|ssl-keygen|setup-|ownership)-[A-Za-z0-9_]+[[:space:]]'
  1167→  } > "$diag_dir/summary.txt" 2>/dev/null || true
  1168→  while IFS=$'\t' read -r name status image; do
  1169→    [ -n "$name" ] || continue
  1170→    docker logs --tail 150 "$name" > "$diag_dir/$name.log" 2>&1 || true
  1171→    if echo "$status" | grep -q 'health'; then
  1172→      docker inspect --format '{{json .State.Health}}' "$name" > "$diag_dir/$name.health.json" 2>/dev/null || true
  1173→    fi
  1174→  done < "$diag_dir/summary.txt"
  1175→  if [ -s "$diag_dir/summary.txt" ]; then
  1176→    msg_info "Состояние контейнеров стека:"
  1177→    sed 's/^/    /' "$diag_dir/summary.txt" || true
  1178→  fi
  1179→  local os_log hint=""
  1180→  os_log="$(ls "$diag_dir"/*opensearch*.log 2>/dev/null | grep -v dashboards | head -1 || true)"
  1181→  if [ -n "$os_log" ] && [ -s "$os_log" ]; then
  1182→    if grep -q 'minimum compatible version\|was created with version' "$os_log"; then
  1183→      hint="OpenSearch не принял данные: в volume есть индексы, созданные версией < 2.0.0 (старые данные событий).
  1184→    Варианты: (а) повторить обновление с --opensearch-data wipe — история событий будет потеряна,
  1185→    настройки правил в PostgreSQL сохраняются; (б) промежуточное обновление на шаблон с OpenSearch 2.x + переиндексация."
  1186→    elif grep -q 'max_map_count' "$os_log"; then
  1187→      hint="На хосте vm.max_map_count < 262144. Исправить: sudo sysctl -w vm.max_map_count=262144;
  1188→    echo 'vm.max_map_count=262144' | sudo tee /etc/sysctl.d/99-clearndr.conf — затем stamusctl compose up -d"
  1189→    elif grep -qiE 'OutOfMemory|Cannot allocate memory' "$os_log" || docker ps -a --format '{{.Status}}' 2>/dev/null | grep -q '137'; then
  1190→      hint="OpenSearch убит по памяти (OOM). Уменьшить heap: stamusctl config set opensearch.memory=1g, затем stamusctl compose up -d"
  1191→    elif grep -qiE 'flood-stage|disk usage exceeded|disk watermark' "$os_log"; then
  1192→      hint="Не хватает места на диске (disk watermark). Освободите место и повторите stamusctl compose up -d"
  1193→    elif grep -qiE 'failed to obtain node lock|NodeLock' "$os_log"; then
  1194→      hint="Данные opensearch заняты другим контейнером/процессом. Проверьте docker ps -a на дубли opensearch"
  1195→    else
  1196→      hint="Причина по журналу opensearch не опознана — смотрите полный журнал в $diag_dir"
  1197→    fi
  1198→    msg_warn "Вероятная причина:
  1199→    $hint"
  1200→  fi
  1201→  msg_ok "Диагностика сохранена: $diag_dir"
  1202→}

  1204→rollback() {
  1205→  local cause="$1"
  1206→  step "ОТКАТ: $cause"
  1207→  [ "$NO_ROLLBACK" -eq 1 ] && { msg_warn "Откат отключён (--no-rollback) — выход."; return 2; }
  1208→  if [ "$DRY_RUN" -eq 1 ]; then
  1209→    msg_info "dry-run: откат — restore config/ из $BACKUP_DIR/config-$TS.tar.gz, затем compose up -d"
  1210→    return 2
  1211→  fi
  1212→  local cfg_archive="$BACKUP_DIR/config-$TS.tar.gz"
  1213→  [ -s "$cfg_archive" ] || { msg_err "Архив config/ для отката не найден: $cfg_archive"; return 2; }
  1214→  stack compose down || true
  1215→  local broken="$INSTANCE_DIR/config.failed-$TS"
  1216→  mv "$CONFIG_DIR" "$broken" && msg_info "Текущий (сломанный) config/ перемещён в $broken"
  1217→  if tar xzf "$cfg_archive" -C "$INSTANCE_DIR"; then
  1218→    msg_ok "config/ восстановлен из архива"
  1219→    if run_prog "Запуск прежнего стека" stack compose up -d; then
  1220→      msg_ok "Команда запуска выполнена"
  1221→    else
  1222→      msg_warn "compose up завершился ошибкой (см. ниже/в логе)"
  1223→    fi
  1224→    # проверяем, что прежний стек действительно поднялся (а не печатаем ОК вслепую)
  1225→    local rb_wait=0 rb_all_up=0 s rb_st
  1226→    while [ "$rb_wait" -lt 120 ]; do
  1227→      rb_all_up=1
  1228→      for s in "${STACK_SERVICES[@]}"; do
  1229→        rb_st="$(docker ps --format '{{.Names}} {{.Status}}' 2>/dev/null \
  1230→          | grep -E -- "-${s}-[A-Za-z0-9_]+ " | head -1 | awk '{print $2}')"
  1231→        [ "$rb_st" = "Up" ] || rb_all_up=0
  1232→      done
  1233→      [ "$rb_all_up" -eq 1 ] && break
  1234→      sleep 10; rb_wait=$((rb_wait + 10))
  1235→    done
  1236→    if [ "$rb_all_up" -eq 1 ]; then
  1237→      msg_ok "Стек поднят в прежней конфигурации ($OLD_STACK_VERSION): сервисы ${STACK_SERVICES[*]} — Up"
  1238→    else
  1239→      msg_warn "Стек поднят в прежней конфигурации ($OLD_STACK_VERSION), но не все сервисы Up за ${rb_wait}с."
  1240→      capture_failure_diagnostics
  1241→      print_emergency_recovery
  1242→    fi
  1243→    cat <<EOF
  1244→  Важно: данные PostgreSQL (volume) НЕ откатывались. Если UI/правила работают некорректно,
  1245→  восстановите БД из дампа:
  1246→    stamusctl compose up -d db
  1247→    zcat $BACKUP_DIR/db-dump-$TS.sql.gz | docker exec -i \$(stamusctl compose ps -q db) psql -U $DBUSER -d postgres
  1248→    stamusctl compose restart
  1249→EOF
  1250→    return 2
  1251→  else
  1252→    msg_err "Не удалось распаковать архив config/ — восстановите вручную из $cfg_archive"
  1253→    return 2
  1254→  fi
  1255→}

  1260→  Частые причины и шаги (подробно: docs/clearndr-update.md, §8.1):

  1266→       docker run --rm -v <том>:/src:ro -v /opt/ClearNDR/backups:/dst alpine tar czf /dst/os-events-backup.tar.gz -C /src .

  1268→       stamusctl compose up -d
  1269→       (история событий теряется; настройки правил в PostgreSQL НЕ теряются)
  1270→EOF
